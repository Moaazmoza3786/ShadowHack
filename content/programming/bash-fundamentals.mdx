# Bash Scripting Fundamentals

Bash is the native language of Linux — and Linux is the native environment of penetration testing. Every pentester needs to be fluent in Bash to automate repetitive tasks, chain tools together, and process output efficiently.

## The Shebang & Basics

```bash
#!/bin/bash
# This line tells the OS to use bash to run this script

# Make it executable
chmod +x script.sh

# Run it
./script.sh
# Or: bash script.sh
```

## Variables

```bash
# Assign (no spaces around =)
TARGET="10.10.10.5"
PORT=80
WORDLIST="/usr/share/wordlists/rockyou.txt"

# Use variables with $
echo "Scanning $TARGET on port $PORT"

# Command substitution — store command output in variable
OPEN_PORTS=$(nmap -p- --open -T4 $TARGET | grep "open" | cut -d'/' -f1)
CURRENT_DATE=$(date +%Y-%m-%d)
HOSTNAME=$(hostname)

# Read user input
read -p "Enter target IP: " TARGET
read -sp "Enter password: " PASSWORD  # -s = silent (no echo)
echo ""  # newline after silent input
```

## Conditionals

```bash
# if/elif/else
if [ $PORT -eq 80 ]; then
    echo "HTTP"
elif [ $PORT -eq 443 ]; then
    echo "HTTPS"
else
    echo "Unknown port"
fi

# String comparison
if [ "$SERVICE" = "ssh" ]; then
    echo "SSH found!"
fi

# File checks
if [ -f "/etc/passwd" ]; then
    echo "File exists"
fi

if [ -d "/tmp" ]; then
    echo "Directory exists"
fi

if [ -r "/etc/shadow" ]; then
    echo "Shadow file is readable — privesc opportunity!"
fi

# Command success check
if nmap -p 80 $TARGET | grep -q "open"; then
    echo "[+] Port 80 is open"
fi
```

## Loops

```bash
# for loop — iterate over list
for PORT in 21 22 80 443 8080 8443; do
    echo "Checking port $PORT..."
    nc -zv $TARGET $PORT 2>&1 | grep -q "succeeded" && echo "[+] $PORT OPEN"
done

# for loop — iterate over range
for i in $(seq 1 254); do
    ping -c 1 -W 1 192.168.1.$i &>/dev/null && echo "[+] 192.168.1.$i is alive"
done

# for loop — iterate over file lines
while IFS= read -r password; do
    # Try each password
    result=$(curl -s -X POST http://$TARGET/login \
        -d "user=admin&pass=$password" | grep -c "Welcome")
    if [ $result -gt 0 ]; then
        echo "[+] Password found: $password"
        break
    fi
done < "$WORDLIST"

# while loop
COUNT=0
while [ $COUNT -lt 10 ]; do
    echo "Attempt $COUNT"
    COUNT=$((COUNT + 1))
done
```

## Functions

```bash
# Define a function
check_port() {
    local ip=$1
    local port=$2
    nc -zv $ip $port 2>&1 | grep -q "succeeded"
    return $?  # 0 = success, 1 = failure
}

# Call it
if check_port "10.10.10.5" 80; then
    echo "[+] Port 80 is open"
fi

# Function with output
get_open_ports() {
    local target=$1
    nmap -p- --open -T4 $target 2>/dev/null | grep "open" | cut -d'/' -f1
}

PORTS=$(get_open_ports "10.10.10.5")
echo "Open ports: $PORTS"
```

## Text Processing — The Holy Trinity

### grep — Search for patterns

```bash
# Find lines containing "password"
grep "password" config.txt

# Case-insensitive
grep -i "password" config.txt

# Show line numbers
grep -n "error" log.txt

# Invert match (show lines NOT matching)
grep -v "nologin" /etc/passwd

# Recursive search
grep -r "api_key" /var/www/

# Count matches
grep -c "Failed" /var/log/auth.log

# Show only matching part
grep -o "192\.168\.[0-9]*\.[0-9]*" log.txt
```

### cut — Extract columns

```bash
# Extract field by delimiter
echo "root:x:0:0:root:/root:/bin/bash" | cut -d':' -f1   # root
echo "root:x:0:0:root:/root:/bin/bash" | cut -d':' -f1,7  # root:/bin/bash

# Extract by character position
echo "10.10.10.5:80" | cut -c1-10   # 10.10.10.

# Get usernames from /etc/passwd
cut -d':' -f1 /etc/passwd
```

### awk — Pattern scanning and processing

```bash
# Print specific column (space-delimited by default)
ps aux | awk '{print $1, $11}'   # user and command

# Print lines where column 3 > 1000
awk -F':' '$3 >= 1000 {print $1}' /etc/passwd   # Non-system users

# Sum a column
awk '{sum += $1} END {print sum}' numbers.txt

# Print lines matching pattern
awk '/FAILED/' /var/log/auth.log
```

## Practical Pentest Scripts

### Ping Sweep

```bash
#!/bin/bash
SUBNET=${1:-"192.168.1"}
echo "[*] Pinging $SUBNET.0/24..."

for i in $(seq 1 254); do
    ping -c 1 -W 1 "$SUBNET.$i" &>/dev/null && \
        echo "[+] $SUBNET.$i is alive" &
done
wait
echo "[*] Done"
```

### Quick Port Check

```bash
#!/bin/bash
TARGET=$1
PORTS="21 22 23 25 53 80 110 139 143 443 445 3306 3389 8080"

echo "[*] Checking common ports on $TARGET"
for PORT in $PORTS; do
    (echo >/dev/tcp/$TARGET/$PORT) 2>/dev/null && \
        echo "[+] $PORT/tcp OPEN" || \
        echo "[-] $PORT/tcp closed"
done
```
