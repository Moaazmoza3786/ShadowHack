# Building a Port Scanner in Python

The best way to learn Python for hacking is to build real tools. A port scanner is the perfect first project — it uses networking, threading, and argument parsing all at once.

## Version 1: Basic Single-Threaded Scanner

```python
#!/usr/bin/env python3
import socket
import sys

def scan_port(ip, port, timeout=1):
    """Returns True if port is open."""
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(timeout)
        result = sock.connect_ex((ip, port))
        sock.close()
        return result == 0
    except Exception:
        return False

def main():
    if len(sys.argv) < 2:
        print(f"Usage: {sys.argv[0]} <target>")
        sys.exit(1)

    target = sys.argv[1]
    print(f"[*] Scanning {target}...")

    for port in range(1, 1025):
        if scan_port(target, port):
            print(f"[+] Port {port}/tcp OPEN")

if __name__ == "__main__":
    main()
```

**Problem:** Scanning 1024 ports one at a time takes ~17 minutes. We need threading.

## Version 2: Multi-Threaded Scanner

```python
#!/usr/bin/env python3
import socket
import threading
import sys
from datetime import datetime

# Thread-safe list for results
open_ports = []
lock = threading.Lock()

def scan_port(ip, port, timeout=0.5):
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(timeout)
        result = sock.connect_ex((ip, port))
        sock.close()
        if result == 0:
            with lock:  # Thread-safe append
                open_ports.append(port)
    except Exception:
        pass

def main():
    target = sys.argv[1] if len(sys.argv) > 1 else "127.0.0.1"
    start_port = int(sys.argv[2]) if len(sys.argv) > 2 else 1
    end_port = int(sys.argv[3]) if len(sys.argv) > 3 else 1024

    print(f"[*] Scanning {target} ports {start_port}-{end_port}")
    start_time = datetime.now()

    threads = []
    for port in range(start_port, end_port + 1):
        t = threading.Thread(target=scan_port, args=(target, port))
        threads.append(t)
        t.start()

        # Limit concurrent threads to avoid overwhelming the target
        if len(threads) >= 100:
            for t in threads:
                t.join()
            threads = []

    # Wait for remaining threads
    for t in threads:
        t.join()

    # Print results
    print(f"\n[+] Open ports on {target}:")
    for port in sorted(open_ports):
        print(f"    {port}/tcp open")

    elapsed = datetime.now() - start_time
    print(f"\n[*] Scan completed in {elapsed}")

if __name__ == "__main__":
    main()
```

## Version 3: Professional Scanner with Banner Grabbing

```python
#!/usr/bin/env python3
import socket
import threading
import argparse
import sys
from datetime import datetime

results = {}
lock = threading.Lock()

# Common port-to-service mapping
SERVICES = {
    21: "FTP", 22: "SSH", 23: "Telnet", 25: "SMTP",
    53: "DNS", 80: "HTTP", 110: "POP3", 143: "IMAP",
    443: "HTTPS", 445: "SMB", 3306: "MySQL", 3389: "RDP",
    5432: "PostgreSQL", 6379: "Redis", 8080: "HTTP-Alt",
    8443: "HTTPS-Alt", 27017: "MongoDB"
}

def grab_banner(sock):
    """Try to grab service banner."""
    try:
        sock.send(b"HEAD / HTTP/1.0\r\n\r\n")
        banner = sock.recv(1024).decode("utf-8", errors="ignore").strip()
        return banner[:100]  # First 100 chars
    except Exception:
        try:
            banner = sock.recv(1024).decode("utf-8", errors="ignore").strip()
            return banner[:100]
        except Exception:
            return ""

def scan_port(ip, port, timeout=1, grab=False):
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(timeout)
        result = sock.connect_ex((ip, port))

        if result == 0:
            banner = grab_banner(sock) if grab else ""
            service = SERVICES.get(port, "unknown")
            with lock:
                results[port] = {"service": service, "banner": banner}
        sock.close()
    except Exception:
        pass

def main():
    parser = argparse.ArgumentParser(description="Python Port Scanner")
    parser.add_argument("target", help="Target IP or hostname")
    parser.add_argument("-p", "--ports", default="1-1024",
                        help="Port range (e.g., 1-1024 or 80,443,8080)")
    parser.add_argument("-t", "--threads", type=int, default=100,
                        help="Number of threads (default: 100)")
    parser.add_argument("-b", "--banner", action="store_true",
                        help="Grab service banners")
    parser.add_argument("--timeout", type=float, default=1.0,
                        help="Connection timeout in seconds")
    args = parser.parse_args()

    # Parse port range
    if "-" in args.ports:
        start, end = map(int, args.ports.split("-"))
        port_list = range(start, end + 1)
    else:
        port_list = [int(p) for p in args.ports.split(",")]

    # Resolve hostname
    try:
        target_ip = socket.gethostbyname(args.target)
    except socket.gaierror:
        print(f"[-] Cannot resolve {args.target}")
        sys.exit(1)

    print(f"\n{'='*50}")
    print(f"  Target: {args.target} ({target_ip})")
    print(f"  Ports:  {args.ports}")
    print(f"  Threads: {args.threads}")
    print(f"{'='*50}\n")

    start_time = datetime.now()
    threads = []

    for port in port_list:
        t = threading.Thread(
            target=scan_port,
            args=(target_ip, port, args.timeout, args.banner)
        )
        threads.append(t)
        t.start()

        if len(threads) >= args.threads:
            for t in threads:
                t.join()
            threads = []

    for t in threads:
        t.join()

    # Display results
    if results:
        print(f"{'PORT':<10} {'SERVICE':<15} {'BANNER'}")
        print("-" * 60)
        for port in sorted(results.keys()):
            info = results[port]
            banner = info['banner'][:40] if info['banner'] else ""
            print(f"{port}/tcp{'':<5} {info['service']:<15} {banner}")
    else:
        print("[-] No open ports found")

    elapsed = datetime.now() - start_time
    print(f"\n[*] Scanned {len(port_list)} ports in {elapsed}")

if __name__ == "__main__":
    main()
```

<Quiz
  question="Why do we use threading.Lock() when appending to the open_ports list?"
  answer="To prevent race conditions — multiple threads writing to the same list simultaneously can corrupt data"
  hint="Without a lock, two threads might try to append at the exact same time, causing data corruption."
/>
